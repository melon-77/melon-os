cmake_minimum_required(VERSION 3.16)
project(melon-welcome CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
add_executable(melon-welcome main.cpp)
target_link_libraries(melon-welcome PRIVATE Qt6::Widgets Qt6::Network)
install(TARGETS melon-welcome RUNTIME DESTINATION bin)
