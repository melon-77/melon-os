# The desktop profile: installed on top of the base system by both installers.
# Since 12 October 2026 (the owner's decision) melon's default desktop is Wayfire, with LXQt's panel, launcher,
# notifications and file manager on top: faster to start and much smaller than Plasma, which stays in the package
# repository (`sudo apk add plasma-desktop plasma-workspace kwin ...`, see README "Plasma"). The 32-bit edition has its
# own list (desktop-packages-x86.txt: the same LXQt on labwc).

eudev dbus elogind polkit linux-pam

mesa mesa-dri mesa-vulkan mesa-va xwayland vulkan-loader

melon-desktop melon-lxqt melon-wayfire wayfire labwc lxqt-session lxqt-wayland-session lxqt-panel lxqt-runner
lxqt-notificationd lxqt-policykit lxqt-powermanagement lxqt-config lxqt-globalkeys lxqt-qtplugin lxqt-themes lxqt-menu-data
pcmanfm-qt qterminal pavucontrol-qt lximage-qt xdg-user-dirs kf6-breeze-icons sddm
# apps
nano
# melon's own games
melon-pinball
# services
pipewire wireplumber networkmanager bluez upower power-profiles-daemon
# runtime pieces nothing links against (loaded by name, QML imports, data files)
xkeyboard-config qt6-qtwayland qt6-qt5compat qt6-qtimageformats qt6-qtsvg hwdata qt6-qttranslations

melon-fonts hicolor-icon-theme shared-mime-info

cryptsetup dosfstools

melon-vm-guest
